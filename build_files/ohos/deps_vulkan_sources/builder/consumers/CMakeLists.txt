cmake_minimum_required(VERSION 3.28)
project(PortableVulkanLibraryConsumers LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_SKIP_RPATH ON)
find_package(Threads REQUIRED)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "HarmonyOS" OR CMAKE_CROSSCOMPILING)
  message(FATAL_ERROR "Actual native HarmonyOS/aarch64 required")
endif()
if(NOT DEFINED PINNED_PREFIX OR NOT EXISTS "${SYSTEM_VULKAN_LOADER}")
  message(FATAL_ERROR "Explicit private prefix and existing system loader required")
endif()
if(VULKAN_CONSUMER STREQUAL "CMAKE")
  find_package(ShaderC CONFIG REQUIRED)
  find_package(SPIRVReflect CONFIG REQUIRED)
  find_package(VulkanHeaders CONFIG REQUIRED)
  find_package(VulkanUtilityLibraries CONFIG REQUIRED)
  find_package(SPIRV-Tools CONFIG REQUIRED)
  find_package(SPIRV-Tools-opt CONFIG REQUIRED)
  find_package(glslang CONFIG REQUIRED)
  find_package(VulkanMemoryAllocator CONFIG REQUIRED)
  set(shader ShaderC::shaderc_combined)
  set(reflect SPIRVReflect::SPIRVReflect)
  set(utility Vulkan::SafeStruct Vulkan::LayerSettings Vulkan::UtilityHeaders)
  set(glslangapis glslang::glslang glslang::glslang-default-resource-limits)
  set(toolsapis SPIRV-Tools-opt SPIRV-Tools-static)
  set(publicapis GPUOpen::VulkanMemoryAllocator)
  get_target_property(_reflect_links SPIRVReflect::SPIRVReflect INTERFACE_LINK_LIBRARIES)
  get_target_property(_reflect_defs SPIRVReflect::SPIRVReflect INTERFACE_COMPILE_DEFINITIONS)
  get_target_property(_vma_includes GPUOpen::VulkanMemoryAllocator INTERFACE_INCLUDE_DIRECTORIES)
  if(NOT "SPIRV-Headers::SPIRV-Headers" IN_LIST _reflect_links OR NOT "SPIRV_REFLECT_USE_SYSTEM_SPIRV_H" IN_LIST _reflect_defs)
    message(FATAL_ERROR "Reflect import lost its own public Headers/macro dependency")
  endif()
  if(NOT "${PINNED_PREFIX}/include" IN_LIST _vma_includes)
    message(FATAL_ERROR "VMA export lacks its own relocated include contract")
  endif()
elseif(VULKAN_CONSUMER STREQUAL "PC")
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SHADER REQUIRED IMPORTED_TARGET shaderc_combined)
  pkg_check_modules(REFLECT REQUIRED IMPORTED_TARGET spirv-reflect)
  pkg_check_modules(UTILITY REQUIRED IMPORTED_TARGET vulkan-utility)
  pkg_check_modules(GLSLANG_API REQUIRED IMPORTED_TARGET glslang)
  pkg_check_modules(TOOLS_API REQUIRED IMPORTED_TARGET SPIRV-Tools-opt)
  pkg_check_modules(PUBLIC REQUIRED IMPORTED_TARGET vma)
  set(shader PkgConfig::SHADER)
  set(reflect PkgConfig::REFLECT)
  set(utility PkgConfig::UTILITY)
  set(glslangapis PkgConfig::GLSLANG_API)
  set(toolsapis PkgConfig::TOOLS_API)
  set(publicapis PkgConfig::PUBLIC)
else()
  message(FATAL_ERROR "VULKAN_CONSUMER must be CMAKE or PC")
endif()
add_executable(library-acceptance integration.cc)
target_link_libraries(library-acceptance PRIVATE ${shader} ${reflect} Threads::Threads)
# Optimizer/validator must come from genuine combined, no extra glslang/Tools here.
add_executable(public-dependency-acceptance public_dependency_link.cc)
target_link_libraries(public-dependency-acceptance PRIVATE ${shader} ${reflect} Threads::Threads)
add_executable(utility-acceptance utility_smoke.cc)
target_link_libraries(utility-acceptance PRIVATE ${utility} Threads::Threads)
add_executable(borrowed-handle-acceptance borrowed_handles.cc)
target_link_libraries(borrowed-handle-acceptance PRIVATE ${utility} Threads::Threads)
add_executable(public-package-acceptance package_apis.cc)
if(VULKAN_CONSUMER STREQUAL "CMAKE")
  # Upstream Vulkan/VMA exports leave the platform selector to the application.
  target_compile_definitions(public-package-acceptance PRIVATE VK_USE_PLATFORM_OHOS)
endif()
target_link_libraries(public-package-acceptance PRIVATE ${publicapis} Threads::Threads)
add_executable(glslang-package-acceptance glslang_api.cc)
target_link_libraries(glslang-package-acceptance PRIVATE ${glslangapis} Threads::Threads)
add_executable(tools-package-acceptance tools_api.cc)
target_link_libraries(tools-package-acceptance PRIVATE ${toolsapis} Threads::Threads)
# Independent immutable copies of Blender's variable-based Find modules.
list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}/../cmake/blender-modules")
set(SHADERC_ROOT_DIR "${PINNED_PREFIX}")
set(VULKAN_ROOT_DIR "${PINNED_PREFIX}")
set(VULKAN_INCLUDE_DIR "${PINNED_PREFIX}/include" CACHE PATH "Complete original headers")
set(VULKAN_LIBRARY "${SYSTEM_VULKAN_LOADER}" CACHE FILEPATH "External actual system loader")
find_package(ShaderC MODULE REQUIRED)
find_package(Vulkan MODULE REQUIRED)
if(NOT SHADERC_LIBRARY STREQUAL "${PINNED_PREFIX}/lib/libshaderc_combined.a" OR NOT VULKAN_LIBRARY STREQUAL SYSTEM_VULKAN_LOADER)
  message(FATAL_ERROR "Blender Find modules selected unexpected archive/loader")
endif()
add_executable(blender-find-acceptance public_dependency_link.cc)
target_include_directories(blender-find-acceptance PRIVATE ${SHADERC_INCLUDE_DIRS})
target_link_libraries(blender-find-acceptance PRIVATE ${SHADERC_LIBRARIES} ${reflect} Threads::Threads)
add_executable(system-loader-acceptance loader_probe.cc)
target_include_directories(system-loader-acceptance PRIVATE ${VULKAN_INCLUDE_DIRS})
target_link_libraries(system-loader-acceptance PRIVATE ${VULKAN_LIBRARIES} ${CMAKE_DL_LIBS})
# Cover combined, Reflect, VUL, actual standalone glslang/Tools closure.
foreach(group combined reflect utility standalone)
  add_library(pic-${group} MODULE pic.cc)
  target_link_options(pic-${group} PRIVATE -Wl,-z,text -Wl,--no-undefined)
  if(group STREQUAL "combined")
    set(whole "${PINNED_PREFIX}/lib/libshaderc_combined.a")
  elseif(group STREQUAL "reflect")
    set(whole "${PINNED_PREFIX}/lib/libspirv-reflect-static.a")
  elseif(group STREQUAL "utility")
    set(whole "${PINNED_PREFIX}/lib/libVulkanSafeStruct.a" "${PINNED_PREFIX}/lib/libVulkanLayerSettings.a")
  else()
    set(whole "${PINNED_PREFIX}/lib/libglslang.a" "${PINNED_PREFIX}/lib/libglslang-default-resource-limits.a" "${PINNED_PREFIX}/lib/libSPIRV-Tools-opt.a" "${PINNED_PREFIX}/lib/libSPIRV-Tools.a")
  endif()
  target_link_libraries(pic-${group} PRIVATE -Wl,--whole-archive ${whole} -Wl,--no-whole-archive Threads::Threads m dl)
endforeach()
