cmake_minimum_required(VERSION 3.28)
project(OHOSCoreAcceptance LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "HarmonyOS" OR CMAKE_CROSSCOMPILING OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "aarch64")
  message(FATAL_ERROR "Actual native HarmonyOS/aarch64 required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_RPATH_USE_ORIGIN ON)
set(CMAKE_JOB_POOLS "compile=2;link=1")
set(CMAKE_JOB_POOL_COMPILE compile)
set(CMAKE_JOB_POOL_LINK link)
try_run(NATIVE_RUN NATIVE_COMPILE
  SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/native_probe.cc"
  CMAKE_FLAGS "-DCMAKE_CXX_STANDARD=20"
  COMPILE_OUTPUT_VARIABLE NATIVE_COMPILE_OUTPUT
  RUN_OUTPUT_VARIABLE NATIVE_RUN_OUTPUT)
file(WRITE "${CMAKE_BINARY_DIR}/native-configure-probe.txt" "system=${CMAKE_SYSTEM_NAME}\nprocessor=${CMAKE_SYSTEM_PROCESSOR}\ncross=${CMAKE_CROSSCOMPILING}\ncompiler=${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}\ncompile=${NATIVE_COMPILE}\nrun=${NATIVE_RUN}\n${NATIVE_COMPILE_OUTPUT}\n${NATIVE_RUN_OUTPUT}")
if(NOT NATIVE_COMPILE OR NOT NATIVE_RUN EQUAL 0)
  message(FATAL_ERROR "Signed native configure probe failed: ${NATIVE_COMPILE_OUTPUT} ${NATIVE_RUN_OUTPUT}")
endif()
message(STATUS "Actual signed configure probe: ${NATIVE_RUN_OUTPUT}")
find_package(Eigen3 CONFIG REQUIRED)
find_package(TBB CONFIG REQUIRED COMPONENTS tbb tbbmalloc)
foreach(component tbb tbbmalloc)
  get_target_property(_core_tbb_type TBB::${component} TYPE)
  if(NOT _core_tbb_type STREQUAL "STATIC_LIBRARY")
    message(FATAL_ERROR "Actual static TBB target required")
  endif()
  get_target_property(_core_tbb_location TBB::${component} IMPORTED_LOCATION_RELEASE)
  if(NOT _core_tbb_location)
    get_target_property(_core_tbb_location TBB::${component} IMPORTED_LOCATION)
  endif()
  get_filename_component(_core_tbb_real "${_core_tbb_location}" REALPATH)
  get_filename_component(_core_tbb_expected "${CORE_DECLARED_TBB_PREFIX}/lib/lib${component}.a" REALPATH)
  if(NOT _core_tbb_real STREQUAL _core_tbb_expected)
    message(FATAL_ERROR "TBB imported target resolves outside declared audited prefix")
  endif()
endforeach()
option(CORE_WITH_EMBREE "Run real Embree consumer after library install" OFF)
option(CORE_WITH_CERES "Run real Ceres consumer after library install" OFF)
option(CORE_WITH_LOGGING "Run real libmv glog and gflags closure" OFF)
enable_testing()
add_executable(native_probe native_probe.cc)
add_test(NAME native_probe COMMAND native_probe)
add_executable(eigen_acceptance eigen_acceptance.cc)
target_link_libraries(eigen_acceptance PRIVATE Eigen3::Eigen TBB::tbb)
target_compile_definitions(eigen_acceptance PRIVATE EIGEN_HAS_TBB)
target_compile_options(eigen_acceptance PRIVATE -UNDEBUG)
add_test(NAME eigen_acceptance COMMAND eigen_acceptance)
if(CORE_WITH_EMBREE)
  find_package(embree 4.4.1 CONFIG REQUIRED)
  add_executable(embree_acceptance embree_acceptance.cc)
  target_compile_options(embree_acceptance PRIVATE -UNDEBUG)
  target_link_libraries(embree_acceptance PRIVATE embree)
  add_test(NAME embree_acceptance COMMAND embree_acceptance)
endif()
if(CORE_WITH_CERES)
  find_package(Ceres CONFIG REQUIRED)
  add_executable(ceres_acceptance ceres_acceptance.cc)
  target_compile_options(ceres_acceptance PRIVATE -UNDEBUG)
  target_link_libraries(ceres_acceptance PRIVATE Ceres::ceres)
  add_test(NAME ceres_acceptance COMMAND ceres_acceptance)
endif()
if(CORE_WITH_LOGGING)
  find_package(gflags CONFIG REQUIRED)
  find_package(glog CONFIG REQUIRED)
  add_executable(logging_acceptance logging_acceptance.cc)
  target_link_libraries(logging_acceptance PRIVATE glog::glog gflags)
  target_compile_options(logging_acceptance PRIVATE -UNDEBUG)
  add_test(NAME logging_acceptance COMMAND logging_acceptance)
endif()
if(CORE_WITH_EMBREE AND CORE_WITH_CERES)
  add_library(core_pic_bridge SHARED pic_bridge.cc)
  target_link_libraries(core_pic_bridge PRIVATE embree Ceres::ceres Eigen3::Eigen TBB::tbb)
  target_link_options(core_pic_bridge PRIVATE -Wl,--no-undefined)
  add_executable(pic_acceptance pic_acceptance.cc)
  target_link_libraries(pic_acceptance PRIVATE core_pic_bridge)
  target_compile_options(pic_acceptance PRIVATE -UNDEBUG)
  set_target_properties(pic_acceptance PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
  add_test(NAME pic_acceptance COMMAND pic_acceptance)
endif()
