cmake_minimum_required(VERSION 3.28)
project(ProvidedNativeOHOSToolkitProbe LANGUAGES C CXX)
if(NOT CMAKE_HOST_SYSTEM_NAME STREQUAL HarmonyOS OR CMAKE_CROSSCOMPILING)
  message(FATAL_ERROR "Actual native HarmonyOS compiler probes required")
endif()
find_package(Threads REQUIRED)
include(CheckCSourceRuns)
check_c_source_runs("#include <sys/mman.h>\n#include <unistd.h>\nint main(){void*p=mmap(0,4096,PROT_READ|PROT_WRITE,MAP_PRIVATE|MAP_ANONYMOUS,-1,0);if(p==MAP_FAILED)return 1;((char*)p)[0]=42;return munmap(p,4096);}" TOOLKIT_REAL_MMAP_RUN)
if(NOT TOOLKIT_REAL_MMAP_RUN)
  message(FATAL_ERROR "Genuine signed mmap native probe failed")
endif()
include("${CMAKE_CURRENT_SOURCE_DIR}/../cmake/native-link-features.cmake")
try_run(TOOLKIT_NATIVE_RUN TOOLKIT_NATIVE_COMPILED SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/native_probe.cc"
  CXX_STANDARD 20 LINK_LIBRARIES Threads::Threads RUN_OUTPUT_VARIABLE TOOLKIT_NATIVE_OUTPUT COMPILE_OUTPUT_VARIABLE TOOLKIT_NATIVE_COMPILE_OUTPUT)
if(NOT TOOLKIT_NATIVE_COMPILED OR NOT TOOLKIT_NATIVE_RUN EQUAL 0)
  message(FATAL_ERROR "Genuine signed native try_run failed: ${TOOLKIT_NATIVE_COMPILE_OUTPUT} ${TOOLKIT_NATIVE_OUTPUT}")
endif()
file(WRITE "${CMAKE_BINARY_DIR}/genuine-native-try-run.txt" "${TOOLKIT_NATIVE_COMPILED};${TOOLKIT_NATIVE_RUN}\n${TOOLKIT_NATIVE_OUTPUT}\n")
foreach(probe native_probe affinity_probe exceptions_rtti_format_error cpp17)
  add_executable(${probe} ${probe}.cc)
  target_compile_features(${probe} PRIVATE cxx_std_20)
  target_link_libraries(${probe} PRIVATE Threads::Threads m dl)
endforeach()
set_property(TARGET cpp17 PROPERTY CXX_STANDARD 17)
set_property(TARGET cpp17 PROPERTY CXX_STANDARD_REQUIRED TRUE)
# Keep this translation unit at17 despite default required20 for other probes.
set_property(TARGET cpp17 PROPERTY COMPILE_FEATURES "cxx_std_17")
add_executable(c17 c17.c)
set_property(TARGET c17 PROPERTY C_STANDARD 17)
add_library(owner_module MODULE owner_module.cc)
target_compile_features(owner_module PRIVATE cxx_std_20)
set_target_properties(owner_module PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_link_options(owner_module PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/owner.exports" "LINKER:-z,text" "LINKER:--no-undefined")
add_executable(module_owner module_owner.cc)
target_link_libraries(module_owner PRIVATE dl)
