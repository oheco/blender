cmake_minimum_required(VERSION 3.20)
project(GltfNativeConsumer LANGUAGES C CXX)
add_executable(gltf_native_acceptance native_acceptance.cc)
set_target_properties(gltf_native_acceptance PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON)
if(GLTF_CONSUMER STREQUAL "CMAKE")
  find_package(draco 1.5.7 EXACT CONFIG REQUIRED PATHS "${GLTF_PREFIX}/share/cmake/draco" NO_DEFAULT_PATH)
  find_package(meshoptimizer 1.1 EXACT CONFIG REQUIRED PATHS "${GLTF_PREFIX}/lib/cmake/meshoptimizer" NO_DEFAULT_PATH)
  find_package(BlenderGltfBridges CONFIG REQUIRED PATHS "${GLTF_PREFIX}/lib/cmake/BlenderGltfBridges" NO_DEFAULT_PATH)
  get_target_property(_dt draco::draco TYPE)
  get_target_property(_mt meshoptimizer::meshoptimizer TYPE)
  if(NOT _dt STREQUAL "STATIC_LIBRARY" OR NOT _mt STREQUAL "STATIC_LIBRARY")
    message(FATAL_ERROR "Static imported libraries required")
  endif()
  target_link_libraries(gltf_native_acceptance PRIVATE draco::draco meshoptimizer::meshoptimizer
    BlenderGltf::bf_intern_draco_bridge BlenderGltf::bf_intern_meshopt_bridge ${CMAKE_DL_LIBS})
elseif(GLTF_CONSUMER STREQUAL "PC")
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(DRACO REQUIRED draco=1.5.7)
  pkg_check_modules(MESHOPT REQUIRED meshoptimizer=1.1)
  pkg_check_modules(BRIDGES REQUIRED BlenderGltfBridges=1)
  target_include_directories(gltf_native_acceptance PRIVATE ${DRACO_INCLUDE_DIRS} ${MESHOPT_INCLUDE_DIRS} ${BRIDGES_INCLUDE_DIRS})
  target_link_directories(gltf_native_acceptance PRIVATE ${DRACO_LIBRARY_DIRS} ${MESHOPT_LIBRARY_DIRS} ${BRIDGES_LIBRARY_DIRS})
  target_compile_options(gltf_native_acceptance PRIVATE ${DRACO_CFLAGS_OTHER} ${MESHOPT_CFLAGS_OTHER} ${BRIDGES_CFLAGS_OTHER})
  target_link_libraries(gltf_native_acceptance PRIVATE ${DRACO_STATIC_LIBRARIES} ${MESHOPT_STATIC_LIBRARIES}
    ${BRIDGES_LIBRARIES} ${CMAKE_DL_LIBS})
else()
  message(FATAL_ERROR "Explicit CMAKE or PC consumer required")
endif()
target_link_options(gltf_native_acceptance PRIVATE "-Wl,-z,defs" "-Wl,-z,text")
