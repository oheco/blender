cmake_minimum_required(VERSION 3.20)
project(BlenderGltfNativeBridges LANGUAGES C CXX)
find_package(draco 1.5.7 EXACT CONFIG REQUIRED PATHS "${GLTF_PREFIX}/share/cmake/draco" NO_DEFAULT_PATH)
find_package(meshoptimizer 1.1 EXACT CONFIG REQUIRED PATHS "${GLTF_PREFIX}/lib/cmake/meshoptimizer" NO_DEFAULT_PATH)
get_target_property(_draco_type draco::draco TYPE)
get_target_property(_meshopt_type meshoptimizer::meshoptimizer TYPE)
if(NOT _draco_type STREQUAL "STATIC_LIBRARY" OR NOT _meshopt_type STREQUAL "STATIC_LIBRARY")
  message(FATAL_ERROR "Actual selected upstream static libraries required")
endif()
foreach(_pair "draco;draco_bridge" "meshopt;meshoptimizer_bridge")
  list(GET _pair 0 _name)
  list(GET _pair 1 _dir)
  set(_src "${GLTF_BLENDER_SOURCE}/intern/${_dir}/intern")
  add_library(bf_intern_${_name}_bridge SHARED "${_src}/common.cpp" "${_src}/encoder.cpp" "${_src}/decoder.cpp")
  set_target_properties(bf_intern_${_name}_bridge PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON
    POSITION_INDEPENDENT_CODE ON INSTALL_RPATH "" BUILD_RPATH "")
  target_link_options(bf_intern_${_name}_bridge PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,-z,defs" "-Wl,-z,text")
  if(_name STREQUAL "draco")
    target_link_libraries(bf_intern_${_name}_bridge PRIVATE draco::draco)
  else()
    target_link_libraries(bf_intern_${_name}_bridge PRIVATE meshoptimizer::meshoptimizer)
  endif()
  install(TARGETS bf_intern_${_name}_bridge EXPORT BlenderGltfBridgesTargets LIBRARY DESTINATION lib)
endforeach()
install(EXPORT BlenderGltfBridgesTargets NAMESPACE BlenderGltf:: DESTINATION lib/cmake/BlenderGltfBridges)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/BlenderGltfBridgesConfig.cmake"
  "include(\"\${CMAKE_CURRENT_LIST_DIR}/BlenderGltfBridgesTargets.cmake\")\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/BlenderGltfBridgesConfig.cmake" DESTINATION lib/cmake/BlenderGltfBridges)
