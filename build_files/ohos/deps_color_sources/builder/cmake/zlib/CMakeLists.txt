# Static-only driver adapted from zlib 1.3.1 CMakeLists.txt; upstream notices
# remain in the pristine source. Native checks below are executed by CMake.
cmake_minimum_required(VERSION 3.28)
project(ColorNativeZlib VERSION 1.3.1 LANGUAGES C)
if(NOT COLOR_ZLIB_SOURCE OR NOT EXISTS "${COLOR_ZLIB_SOURCE}/zlib.h")
  message(FATAL_ERROR "Complete pinned COLOR_ZLIB_SOURCE required")
endif()
include(CheckTypeSize)
include(CheckFunctionExists)
include(CheckIncludeFile)
check_include_file(sys/types.h HAVE_SYS_TYPES_H)
check_include_file(stdint.h HAVE_STDINT_H)
check_include_file(stddef.h HAVE_STDDEF_H)
set(CMAKE_REQUIRED_DEFINITIONS -D_LARGEFILE64_SOURCE=1)
foreach(feature HAVE_SYS_TYPES_H HAVE_STDINT_H HAVE_STDDEF_H)
  if(${feature})
    list(APPEND CMAKE_REQUIRED_DEFINITIONS -D${feature})
  endif()
endforeach()
check_type_size(off64_t OFF64_T)
set(CMAKE_REQUIRED_DEFINITIONS)
check_function_exists(fseeko HAVE_FSEEKO)
check_include_file(unistd.h Z_HAVE_UNISTD_H)
configure_file("${COLOR_ZLIB_SOURCE}/zconf.h.cmakein" "${CMAKE_CURRENT_BINARY_DIR}/zconf.h" @ONLY)
set(_sources adler32.c compress.c crc32.c deflate.c gzclose.c gzlib.c gzread.c gzwrite.c
  inflate.c infback.c inftrees.c inffast.c trees.c uncompr.c zutil.c)
set(_headers zlib.h crc32.h deflate.h gzguts.h inffast.h inffixed.h inflate.h inftrees.h trees.h zutil.h)
# Build copies resolve the genuinely configured zconf.h without renaming or
# mutating any original complete source file.
foreach(file IN LISTS _sources _headers)
  configure_file("${COLOR_ZLIB_SOURCE}/${file}" "${CMAKE_CURRENT_BINARY_DIR}/${file}" COPYONLY)
endforeach()
list(TRANSFORM _sources PREPEND "${CMAKE_CURRENT_BINARY_DIR}/")
add_library(zlibstatic STATIC ${_sources})
set_target_properties(zlibstatic PROPERTIES OUTPUT_NAME z POSITION_INDEPENDENT_CODE ON)
target_include_directories(zlibstatic PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}>" "$<INSTALL_INTERFACE:include>")
if(HAVE_OFF64_T)
  target_compile_definitions(zlibstatic PRIVATE _LARGEFILE64_SOURCE=1)
endif()
if(NOT HAVE_FSEEKO)
  target_compile_definitions(zlibstatic PRIVATE NO_FSEEKO)
endif()
install(TARGETS zlibstatic ARCHIVE DESTINATION lib)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/zconf.h" "${COLOR_ZLIB_SOURCE}/zlib.h" DESTINATION include)
