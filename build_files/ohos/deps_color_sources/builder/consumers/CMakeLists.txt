cmake_minimum_required(VERSION 3.28)
project(HarmonyOSColorAcceptance LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_SKIP_RPATH ON)
set(COLOR_CONSUMER CMAKE CACHE STRING "CMAKE or PC static metadata")
if(COLOR_CONSUMER STREQUAL "PC")
  set(PKG_CONFIG_ARGN --static)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(COLOR REQUIRED IMPORTED_TARGET OpenEXR OpenImageIO OpenColorIO Imath)
  pkg_check_modules(TBB_NATIVE REQUIRED IMPORTED_TARGET tbb tbbmalloc)
  set(tbb_targets PkgConfig::TBB_NATIVE)
  set(color_targets PkgConfig::COLOR)
elseif(COLOR_CONSUMER STREQUAL "CMAKE")
  find_package(OpenEXR 3.4.10 EXACT CONFIG REQUIRED)
  find_package(OpenImageIO 3.1.13.1 EXACT CONFIG REQUIRED)
  find_package(OpenColorIO 2.5.0 EXACT CONFIG REQUIRED)
  find_package(Imath 3.2.2 EXACT CONFIG REQUIRED)
  find_package(TBB 2022.3.0 EXACT CONFIG REQUIRED)
  set(tbb_targets TBB::tbb TBB::tbbmalloc)
  set(color_targets OpenEXR::OpenEXR OpenImageIO::OpenImageIO OpenImageIO::OpenImageIO_Util OpenColorIO::OpenColorIO Imath::Imath)
else()
  message(FATAL_ERROR "COLOR_CONSUMER must be CMAKE or PC")
endif()
add_executable(color_acceptance color_acceptance.cpp)
target_link_libraries(color_acceptance PRIVATE ${color_targets})
add_executable(half_texture_acceptance half_texture_acceptance.cpp)
target_link_libraries(half_texture_acceptance PRIVATE ${color_targets})
# Link a real shared PIC consumer from the same functional implementation.
# This resolves actual dependency objects under -z,text and is never dlopened;
# duplicate static C++ runtime ownership belongs to parent integration.
add_library(color_pic MODULE color_acceptance.cpp)
target_link_libraries(color_pic PRIVATE ${color_targets})
target_link_options(color_pic PRIVATE "-Wl,-z,text")

add_executable(tbb_acceptance tbb_acceptance.cc)
target_compile_options(tbb_acceptance PRIVATE -UNDEBUG)
target_link_libraries(tbb_acceptance PRIVATE ${tbb_targets})
