cmake_minimum_required(VERSION 3.28)
project(ColorNativePrerequisites LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(native_probe ../consumers/native_probe.cc)
target_link_libraries(native_probe PRIVATE Threads::Threads)
add_executable(affinity_probe ../consumers/affinity_probe.cc)
target_link_libraries(affinity_probe PRIVATE Threads::Threads)
enable_testing()
add_test(NAME actual_native_sdk COMMAND native_probe)
add_test(NAME actual_affinity_read_restrict_restore COMMAND affinity_probe)
set_tests_properties(actual_native_sdk actual_affinity_read_restrict_restore PROPERTIES TIMEOUT 30)
