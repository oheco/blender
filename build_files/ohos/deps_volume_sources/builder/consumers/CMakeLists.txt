cmake_minimum_required(VERSION 3.28)
project(HarmonyOSVolumeAcceptance LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_SKIP_RPATH ON)
set(VOLUME_CONSUMER CMAKE CACHE STRING "CMAKE or PC")
find_package(Threads REQUIRED)
if(VOLUME_CONSUMER STREQUAL "PC")
  set(PKG_CONFIG_ARGN --static)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(VOLUME REQUIRED IMPORTED_TARGET openvdb nanovdb)
  pkg_check_modules(FFT REQUIRED IMPORTED_TARGET fftw3_threads fftw3f_threads)
  pkg_check_modules(TBB_NATIVE REQUIRED IMPORTED_TARGET tbb tbbmalloc)
  set(volume_target PkgConfig::VOLUME)
  set(fft_targets PkgConfig::FFT)
  set(tbb_targets PkgConfig::TBB_NATIVE)
elseif(VOLUME_CONSUMER STREQUAL "CMAKE")
  find_package(OpenVDB 13.0.0 EXACT CONFIG REQUIRED)
  find_package(FFTW3 3.3.10 EXACT CONFIG REQUIRED)
  find_package(FFTW3f 3.3.10 EXACT CONFIG REQUIRED)
  find_package(TBB 2022.3.0 EXACT CONFIG REQUIRED)
  set(volume_target OpenVDB::openvdb OpenVDB::nanovdb)
  set(fft_targets FFTW3::fftw3_threads FFTW3::fftw3f_threads)
  set(tbb_targets TBB::tbb TBB::tbbmalloc)
else()
  message(FATAL_ERROR "VOLUME_CONSUMER must be CMAKE or PC")
endif()
add_executable(volume_acceptance volume_acceptance.cpp)
target_link_libraries(volume_acceptance PRIVATE ${volume_target})
add_executable(fftw_acceptance fftw_acceptance.cpp)
target_link_libraries(fftw_acceptance PRIVATE ${fft_targets})
add_executable(tbb_acceptance tbb_acceptance.cc)
target_compile_options(tbb_acceptance PRIVATE -UNDEBUG)
target_link_libraries(tbb_acceptance PRIVATE ${tbb_targets})
add_executable(nanovdb_stream_regression nanovdb_stream_regression.cpp)
target_compile_options(nanovdb_stream_regression PRIVATE -O1)
target_link_libraries(nanovdb_stream_regression PRIVATE ${volume_target})
add_library(volume_pic MODULE pic.cpp)
target_link_libraries(volume_pic PRIVATE "-Wl,--whole-archive" ${volume_target} ${fft_targets} ${tbb_targets} "-Wl,--no-whole-archive")
target_link_options(volume_pic PRIVATE "-Wl,-z,text")
# Module is signed and audited but never dlopened: separate static C++ runtime
# ownership remains integration work. Native executables are run by builder.
