cmake_minimum_required(VERSION 3.28)
project(BasePortableConsumers LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_FIND_PACKAGE_PREFER_CONFIG ON)
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY OFF)
find_package(Threads REQUIRED)
if(NOT MODE MATCHES "^(CMAKE|PC)$")
  message(FATAL_ERROR "Explicit CMAKE/PC mode required")
endif()
set(_packages zlib libzstd libjpeg libturbojpeg libpng libbrotlicommon libbrotlidec libbrotlienc freetype2 harfbuzz harfbuzz-subset fribidi fmt Imath)
set(_targets ZLIB::ZLIB zstd::libzstd_static libjpeg-turbo::jpeg-static libjpeg-turbo::turbojpeg-static PNG::png_static Brotli::common Brotli::dec Brotli::enc Freetype::Freetype harfbuzz::harfbuzz harfbuzz::subset FriBidi::FriBidi fmt::fmt Imath::Imath)
set(_macros ZLIB ZSTD JPEG TURBOJPEG PNG BROTLICOMMON BROTLIDEC BROTLIENC FREETYPE HARFBUZZ HBSUBSET FRIBIDI FMT IMATH)
if(MODE STREQUAL CMAKE)
  set(_configs ZLIB zstd libjpeg-turbo libjpeg-turbo PNG Brotli Brotli Brotli freetype harfbuzz harfbuzz FriBidi fmt Imath)
  if(ONLY_PACKAGE)
    list(FIND _packages "${ONLY_PACKAGE}" _only_index)
    if(_only_index LESS 0)
      message(FATAL_ERROR "Unknown standalone public package")
    endif()
    list(GET _configs ${_only_index} package)
    find_package(${package} CONFIG REQUIRED)
  else()
    foreach(package ZLIB zstd libjpeg-turbo PNG Brotli freetype harfbuzz FriBidi fmt Imath)
      find_package(${package} CONFIG REQUIRED)
    endforeach()
  endif()
else()
  # Static flags are captured via newly source-built pkgconf, decoded before
  # JSON/CMake escaping, and recorded in PC_FLAGS_FILE; no manual closure.
  include("${PC_FLAGS_FILE}")
endif()
set(_all_interfaces)
foreach(index RANGE 0 13)
  list(GET _packages ${index} package)
  list(GET _targets ${index} target)
  list(GET _macros ${index} macro)
  if(ONLY_PACKAGE AND NOT package STREQUAL ONLY_PACKAGE)
    continue()
  endif()
  string(REPLACE "-" "_" id "${package}")
  add_executable(package_${id} package_api.cc)
  target_compile_definitions(package_${id} PRIVATE PK_${macro}=1)
  if(MODE STREQUAL CMAKE)
    if(NOT TARGET ${target})
      message(FATAL_ERROR "Required public import missing: ${target}")
    endif()
    target_link_libraries(package_${id} PRIVATE ${target})
    list(APPEND _all_interfaces ${target})
  else()
    add_library(pc_${id} INTERFACE)
    target_compile_options(pc_${id} INTERFACE ${PC_${id}_CFLAGS})
    # Search paths are typed directories; raw -L link items bypass generator escaping.
    foreach(_pc_link_item IN LISTS PC_${id}_LIBS)
      if(_pc_link_item MATCHES "^-L(.+)$")
        target_link_directories(pc_${id} INTERFACE "${CMAKE_MATCH_1}")
      else()
        target_link_libraries(pc_${id} INTERFACE "${_pc_link_item}")
      endif()
    endforeach()
    target_link_libraries(package_${id} PRIVATE pc_${id})
    list(APPEND _all_interfaces pc_${id})
  endif()
  add_test(NAME package_${id} COMMAND package_${id})
endforeach()
if(ONLY_PACKAGE)
  if(ONLY_PACKAGE STREQUAL Imath)
    add_executable(imath_config_alias imath_config_alias.cc)
    target_link_libraries(imath_config_alias PRIVATE Imath::Config)
  endif()
  return()
endif()
# Required real COVER regression; acceptance.py executes it in both modes and migration.
add_executable(zstd-cover-concurrent zstd_cover_concurrent.cc)
if(MODE STREQUAL CMAKE)
  target_link_libraries(zstd-cover-concurrent PRIVATE zstd::libzstd_static Threads::Threads)
else()
  target_link_libraries(zstd-cover-concurrent PRIVATE pc_libzstd Threads::Threads)
endif()
add_test(NAME sourcebuilt_cover_concurrent COMMAND zstd-cover-concurrent --workers=2 --iterations=6 --codec-workers=2 --optimizer-threads=2 --notification-level=0)
add_executable(native-acceptance native-acceptance.cpp)
target_compile_definitions(native-acceptance PRIVATE HAVE_HARFBUZZ=1 HAVE_FRIBIDI=1 HAVE_FMT=1 HAVE_IMATH=1)
target_link_libraries(native-acceptance PRIVATE ${_all_interfaces})
add_executable(text-layout text_layout.cc)
if(MODE STREQUAL CMAKE)
  target_link_libraries(text-layout PRIVATE harfbuzz::harfbuzz FriBidi::FriBidi)
else()
  target_link_libraries(text-layout PRIVATE pc_harfbuzz pc_fribidi)
endif()
# Image/text/module APIs keep allocations and C++ state in their owner module.
add_library(base_bridge MODULE bridge.cc)
set_target_properties(base_bridge PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
target_link_options(base_bridge PRIVATE "-Wl,-z,text" "-Wl,--no-undefined" "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/bridge.exports")
# Whole archives exercise every original member, separately from actual public
# interface consumers which must prove static closure without manual additions.
if(NOT WHOLE_ARCHIVES)
  message(FATAL_ERROR "Explicit canonical source-built archives required")
endif()
target_link_libraries(base_bridge PRIVATE "-Wl,--whole-archive" ${WHOLE_ARCHIVES} "-Wl,--no-whole-archive" ${_all_interfaces})
add_executable(module-owner module_owner.cc)
target_link_libraries(module-owner PRIVATE dl Threads::Threads)
add_test(NAME sourcebuilt_module_owner COMMAND module-owner "$<TARGET_FILE:base_bridge>" "${PINNED_FONT}")
enable_testing()
foreach(target native-acceptance text-layout zstd-cover-concurrent base_bridge module-owner)
  get_target_property(includes ${target} INCLUDE_DIRECTORIES)
  get_target_property(links ${target} LINK_LIBRARIES)
  file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/resolved-public-interfaces.txt" "${target} includes=${includes}\nlinks=${links}\n")
endforeach()
