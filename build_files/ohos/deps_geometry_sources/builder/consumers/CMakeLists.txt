cmake_minimum_required(VERSION 3.28)
project(GeometryAcceptance LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "HarmonyOS" OR CMAKE_CROSSCOMPILING)
  message(FATAL_ERROR "Actual native HarmonyOS acceptance only")
endif()
find_package(Threads REQUIRED)
if(GEOMETRY_CONSUMER STREQUAL "CMAKE")
  find_package(TBB 2022.3.0 EXACT CONFIG REQUIRED PATHS "${GEOMETRY_PREFIX}/lib/cmake/TBB" NO_DEFAULT_PATH)
  find_package(OpenSubdiv 3.7.0 EXACT CONFIG REQUIRED PATHS "${GEOMETRY_PREFIX}/lib/cmake/OpenSubdiv" NO_DEFAULT_PATH)
  find_package(GMP 6.3.0 EXACT CONFIG REQUIRED PATHS "${GEOMETRY_PREFIX}/lib/cmake/GMP" NO_DEFAULT_PATH)
  find_package(manifold 3.5.2 EXACT CONFIG REQUIRED PATHS "${GEOMETRY_PREFIX}/lib/cmake/manifold" NO_DEFAULT_PATH)
  set(TBB_LINK TBB::tbb TBB::tbbmalloc)
  set(OSD_LINK OpenSubdiv::osdGPU_static OpenSubdiv::osdCPU_static)
  set(GMP_LINK GMP::gmpxx GMP::gmp)
  set(MANIFOLD_LINK manifold::manifold)
elseif(GEOMETRY_CONSUMER STREQUAL "PC")
  find_package(PkgConfig REQUIRED)
  set(PKG_CONFIG_USE_CMAKE_PREFIX_PATH FALSE)
  set(PKG_CONFIG_ARGN --static)
  pkg_check_modules(TBB_PC REQUIRED IMPORTED_TARGET tbb=2022.3.0 tbbmalloc=2022.3.0)
  pkg_check_modules(OSD_PC REQUIRED IMPORTED_TARGET opensubdiv=3.7.0)
  pkg_check_modules(GMP_PC REQUIRED IMPORTED_TARGET gmpxx=6.3.0 gmp=6.3.0)
  pkg_check_modules(MANIFOLD_PC REQUIRED IMPORTED_TARGET manifold=3.5.2)
  set(TBB_LINK PkgConfig::TBB_PC)
  set(OSD_LINK PkgConfig::OSD_PC)
  set(GMP_LINK PkgConfig::GMP_PC)
  set(MANIFOLD_LINK PkgConfig::MANIFOLD_PC)
else()
  message(FATAL_ERROR "Select real CMAKE or PC consumer")
endif()
# Exercise sealed byte-identical Blender find modules against this same prefix.
list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/blender-find")
set(GMP_ROOT_DIR "${GEOMETRY_PREFIX}")
set(OPENSUBDIV_ROOT_DIR "${GEOMETRY_PREFIX}")
include("${CMAKE_CURRENT_SOURCE_DIR}/blender-find/FindGMP.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/blender-find/FindOpenSubdiv.cmake")
if(NOT GMP_FOUND OR NOT OPENSUBDIV_FOUND)
  message(FATAL_ERROR "Sealed Blender find modules did not resolve geometry")
endif()
foreach(variable GMP_LIBRARY GMPXX_LIBRARY OPENSUBDIV_OSDCPU_LIBRARY OPENSUBDIV_OSDGPU_LIBRARY)
  if(NOT "${${variable}}" MATCHES "^${GEOMETRY_PREFIX}/lib/")
    message(FATAL_ERROR "Blender ${variable} escaped canonical geometry prefix")
  endif()
endforeach()
# ShaderC is an explicitly selected, already accepted external source closure.
# Only the real combined archive is linked; no duplicate Tools/glslang archive.
add_library(Geometry::accepted_shaderc STATIC IMPORTED)
set_target_properties(Geometry::accepted_shaderc PROPERTIES
  IMPORTED_LOCATION "${GEOMETRY_SHADER_PREFIX}/lib/libshaderc_combined.a"
  INTERFACE_INCLUDE_DIRECTORIES "${GEOMETRY_SHADER_PREFIX}/include"
  INTERFACE_LINK_LIBRARIES "Threads::Threads;dl;m")
add_executable(tbb_acceptance tbb_acceptance.cc)
target_link_libraries(tbb_acceptance PRIVATE ${TBB_LINK})
add_executable(opensubdiv_acceptance opensubdiv_acceptance.cc)
target_link_libraries(opensubdiv_acceptance PRIVATE ${OSD_LINK})
add_executable(gmp_acceptance gmp_acceptance.cc)
target_link_libraries(gmp_acceptance PRIVATE ${GMP_LINK})
add_executable(manifold_acceptance manifold_acceptance.cc)
target_link_libraries(manifold_acceptance PRIVATE ${MANIFOLD_LINK})
add_executable(glsl_acceptance glsl_acceptance.cc)
target_link_libraries(glsl_acceptance PRIVATE ${OSD_LINK} Geometry::accepted_shaderc)
add_library(geometry_pic MODULE pic.cpp)
target_link_libraries(geometry_pic PRIVATE ${TBB_LINK} ${OSD_LINK} ${GMP_LINK} ${MANIFOLD_LINK})
target_link_options(geometry_pic PRIVATE "-Wl,-z,text" "-Wl,--no-undefined")
foreach(target tbb_acceptance opensubdiv_acceptance gmp_acceptance manifold_acceptance glsl_acceptance geometry_pic)
  target_compile_options(${target} PRIVATE -UNDEBUG)
endforeach()
